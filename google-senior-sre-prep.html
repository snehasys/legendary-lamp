Solution Template: Streaming Log Parser

#!/usr/bin/env python3
"""Parse a multi-GB log file with bounded memory."""
import re
from collections import Counter
import heapq

def parse_logs(filepath, top_n=10):
    pattern = re.compile(
        r'(?P<ip>\S+) .* "(?P<method>\w+) (?P<path>\S+) .*" (?P<status>\d{3}) (?P<size>\d+)'
    )
    error_counts = Counter()

    # Stream line-by-line — never loads full file into memory
    with open(filepath, 'r', buffering=8192) as f:
        for line in f:  # iterator, O(1) memory per line
            match = pattern.search(line)
            if not match:
                continue  # skip malformed lines gracefully
            status = int(match.group('status'))
            if 500 <= status < 600:
                endpoint = match.group('path').split('?')[0]
                error_counts[endpoint] += 1

    # Top-N using heap — O(n log k) instead of full sort
    return heapq.nlargest(top_n, error_counts.items(), key=lambda x: x[1])

Solution Template: Concurrent Health Checker

#!/usr/bin/env python3
"""Check health of N servers with concurrency + rate limiting."""
import asyncio
import aiohttp
from dataclasses import dataclass, field
from typing import List

@dataclass
class HealthResult:
    host: str
    healthy: bool
    latency_ms: float = 0
    error: str = ""

async def check_health(
    hosts: List[str],
    timeout_sec: float = 5.0,
    max_concurrent: int = 50,
    requests_per_sec: int = 50
) -> List[HealthResult]:
    results = []
    semaphore = asyncio.Semaphore(max_concurrent)
    rate_limiter = asyncio.Semaphore(requests_per_sec)

    async def refill_tokens():
        while True:
            await asyncio.sleep(1.0 / requests_per_sec)
            try: rate_limiter.release()
            except ValueError: pass

    async def check_one(session, host):
        await rate_limiter.acquire()
        async with semaphore:
            try:
                start = asyncio.get_event_loop().time()
                async with session.get(
                    f"http://{host}/healthz",
                    timeout=aiohttp.ClientTimeout(total=timeout_sec)
                ) as resp:
                    latency = (asyncio.get_event_loop().time() - start) * 1000
                    results.append(HealthResult(host=host, healthy=(resp.status == 200), latency_ms=latency))
            except (asyncio.TimeoutError, aiohttp.ClientError) as e:
                results.append(HealthResult(host=host, healthy=False, error=str(e)))

    async with aiohttp.ClientSession() as session:
        refiller = asyncio.create_task(refill_tokens())
        await asyncio.gather(*[check_one(session, h) for h in hosts])
        refiller.cancel()
    return results

Solution Template: Token Bucket Rate Limiter

"""Thread-safe token bucket rate limiter."""
import time, threading

class TokenBucket:
    def __init__(self, rate: float, burst: int):
        self.rate = rate        # tokens per second
        self.burst = burst      # max tokens (burst capacity)
        self.tokens = float(burst)
        self.last_refill = time.monotonic()
        self.lock = threading.Lock()

    def _refill(self):
        now = time.monotonic()
        self.tokens = min(self.burst, self.tokens + (now - self.last_refill) * self.rate)
        self.last_refill = now

    def acquire(self, tokens: int = 1) -> bool:
        with self.lock:
            self._refill()
            if self.tokens >= tokens:
                self.tokens -= tokens
                return True
            return False

    def wait(self, tokens: int = 1):
        while not self.acquire(tokens):
            time.sleep(1.0 / self.rate)

Pattern 1: Retry with Exponential Backoff & Jitter

Every network call in production needs retry logic. But naive retries cause thundering herds. The solution: exponential backoff with full jitter.

"""Production-grade retry decorator with exponential backoff + jitter."""
import time, random, functools, logging
from typing import Tuple, Type

logger = logging.getLogger(__name__)

def retry(
    max_attempts: int = 3,
    base_delay: float = 1.0,
    max_delay: float = 60.0,
    exceptions: Tuple[Type[Exception], ...] = (Exception,),
    backoff_factor: float = 2.0
):
    """Decorator: retries with exponential backoff + full jitter."""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    if attempt == max_attempts - 1:
                        logger.error(f"{func.__name__} failed after {max_attempts} attempts: {e}")
                        raise
                    # Exponential backoff with full jitter
                    delay = min(max_delay, base_delay * (backoff_factor ** attempt))
                    jittered = random.uniform(0, delay)  # FULL jitter, not just additive
                    logger.warning(f"{func.__name__} attempt {attempt+1} failed: {e}. "
                                   f"Retrying in {jittered:.2f}s")
                    time.sleep(jittered)
        return wrapper
    return decorator

# Usage:
@retry(max_attempts=5, base_delay=0.5, exceptions=(ConnectionError, TimeoutError))
def fetch_config(url):
    response = requests.get(url, timeout=10)
    response.raise_for_status()
    return response.json()
Why Full Jitter?
AWS's analysis showed that full jitter (random between 0 and calculated delay) reduces total completion time by ~40% compared to equal jitter or no jitter. It decorrelates retry storms across clients.

Pattern 2: Circuit Breaker

When a dependency is failing, stop hammering it. A circuit breaker "opens" after N failures, short-circuits all calls for a cooldown period, then "half-opens" to test recovery.

"""Circuit breaker pattern — prevents cascading failures."""
import time, threading
from enum import Enum

class State(Enum):
    CLOSED = "closed"       # normal operation
    OPEN = "open"           # failing, reject all calls
    HALF_OPEN = "half_open" # testing if dependency recovered

class CircuitBreaker:
    def __init__(self, failure_threshold=5, recovery_timeout=30, half_open_max=3):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_max = half_open_max
        self.state = State.CLOSED
        self.failure_count = 0
        self.last_failure_time = 0
        self.half_open_calls = 0
        self.lock = threading.Lock()

    def call(self, func, *args, **kwargs):
        with self.lock:
            if self.state == State.OPEN:
                if time.time() - self.last_failure_time > self.recovery_timeout:
                    self.state = State.HALF_OPEN
                    self.half_open_calls = 0
                else:
                    raise CircuitOpenError(f"Circuit open, retry after {self.recovery_timeout}s")

            if self.state == State.HALF_OPEN and self.half_open_calls >= self.half_open_max:
                raise CircuitOpenError("Half-open limit reached")

        try:
            result = func(*args, **kwargs)
            with self.lock:
                if self.state == State.HALF_OPEN:
                    self.state = State.CLOSED  # recovered!
                self.failure_count = 0
            return result
        except Exception as e:
            with self.lock:
                self.failure_count += 1
                self.last_failure_time = time.time()
                if self.state == State.HALF_OPEN or self.failure_count >= self.failure_threshold:
                    self.state = State.OPEN
            raise

class CircuitOpenError(Exception):
    pass

Pattern 3: Graceful Shutdown Handler

Production services must handle SIGTERM gracefully — finish in-flight work, close connections, flush buffers, then exit cleanly. Kubernetes sends SIGTERM 30s before SIGKILL.

"""Graceful shutdown with signal handling and cleanup."""
import signal, sys, threading, logging, time

logger = logging.getLogger(__name__)

class GracefulShutdown:
    def __init__(self, timeout: float = 25.0):
        self.timeout = timeout
        self.shutdown_event = threading.Event()
        self.cleanup_tasks = []
        signal.signal(signal.SIGTERM, self._handle_signal)
        signal.signal(signal.SIGINT, self._handle_signal)

    def _handle_signal(self, signum, frame):
        sig_name = signal.Signals(signum).name
        logger.info(f"Received {sig_name}, initiating graceful shutdown...")
        self.shutdown_event.set()

    def register_cleanup(self, func, *args):
        """Register a cleanup function to run on shutdown."""
        self.cleanup_tasks.append((func, args))

    @property
    def should_stop(self) -> bool:
        return self.shutdown_event.is_set()

    def wait_for_shutdown(self):
        """Block until shutdown signal received, then run cleanup."""
        self.shutdown_event.wait()
        logger.info(f"Running {len(self.cleanup_tasks)} cleanup tasks...")
        deadline = time.time() + self.timeout
        for func, args in self.cleanup_tasks:
            if time.time() > deadline:
                logger.error("Cleanup timeout exceeded, forcing exit")
                break
            try:
                func(*args)
            except Exception as e:
                logger.error(f"Cleanup task failed: {e}")
        logger.info("Shutdown complete")
        sys.exit(0)

# Usage:
shutdown = GracefulShutdown(timeout=25)
shutdown.register_cleanup(db_pool.close)
shutdown.register_cleanup(flush_metrics)

while not shutdown.should_stop:
    process_next_item()  # your main loop

Pattern 4: Implement tail -f with Log Rotation Detection

A classic SRE interview question. You must handle: file rotation (inode changes), file truncation, and new lines appearing. This tests your understanding of file descriptors and inodes.

"""Implement tail -f that survives log rotation."""
import os, time, stat

def tail_follow(filepath, poll_interval=0.1):
    """Yield new lines from a file, surviving rotation and truncation."""
    while not os.path.exists(filepath):
        time.sleep(poll_interval)

    fd = open(filepath, 'r')
    fd.seek(0, os.SEEK_END)  # start at end of file
    current_inode = os.fstat(fd.fileno()).st_ino
    current_pos = fd.tell()

    while True:
        line = fd.readline()
        if line:
            yield line.rstrip('\n')
            current_pos = fd.tell()
        else:
            # No new data — check for rotation or truncation
            try:
                st = os.stat(filepath)
            except FileNotFoundError:
                # File deleted, wait for recreation
                time.sleep(poll_interval)
                continue

            if st.st_ino != current_inode:
                # ROTATION: file was replaced (new inode)
                fd.close()
                fd = open(filepath, 'r')
                current_inode = os.fstat(fd.fileno()).st_ino
                current_pos = 0
            elif st.st_size < current_pos:
                # TRUNCATION: file was truncated (same inode, smaller)
                fd.seek(0)
                current_pos = 0
            else:
                time.sleep(poll_interval)

# Usage:
for line in tail_follow('/var/log/app.log'):
    if 'ERROR' in line:
        alert(line)

Pattern 5: Connection Pool with Health Checking

"""Connection pool with periodic health checks and eviction."""
import threading, time, queue
from contextlib import contextmanager

class ConnectionPool:
    def __init__(self, factory, max_size=10, health_interval=30, max_idle=300):
        self.factory = factory        # callable that creates a connection
        self.max_size = max_size
        self.health_interval = health_interval
        self.max_idle = max_idle
        self.pool = queue.Queue(maxsize=max_size)
        self.size = 0
        self.lock = threading.Lock()
        self._start_health_checker()

    def _start_health_checker(self):
        t = threading.Thread(target=self._health_loop, daemon=True)
        t.start()

    def _health_loop(self):
        while True:
            time.sleep(self.health_interval)
            self._evict_stale()

    def _evict_stale(self):
        """Remove connections that are idle too long or unhealthy."""
        healthy = []
        while not self.pool.empty():
            try:
                conn, last_used = self.pool.get_nowait()
                if time.time() - last_used > self.max_idle:
                    conn.close()
                    with self.lock: self.size -= 1
                elif not conn.is_healthy():
                    conn.close()
                    with self.lock: self.size -= 1
                else:
                    healthy.append((conn, last_used))
            except queue.Empty:
                break
        for item in healthy:
            self.pool.put(item)

    @contextmanager
    def acquire(self, timeout=5.0):
        """Context manager: get a connection, return it when done."""
        conn = self._get_connection(timeout)
        try:
            yield conn
        except Exception:
            # Don't return broken connections to pool
            conn.close()
            with self.lock: self.size -= 1
            raise
        else:
            self.pool.put((conn, time.time()))

    def _get_connection(self, timeout):
        try:
            conn, _ = self.pool.get(timeout=0.1)
            if conn.is_healthy():
                return conn
            conn.close()
            with self.lock: self.size -= 1
        except queue.Empty:
            pass
        with self.lock:
            if self.size < self.max_size:
                self.size += 1
                return self.factory()
        # Pool exhausted, wait
        conn, _ = self.pool.get(timeout=timeout)
        return conn

Pattern 6: Sliding Window Counter (Anomaly Detection)

"""Memory-bounded sliding window for real-time anomaly detection."""
from collections import deque
import time

class SlidingWindowCounter:
    """Count events in a sliding time window with O(1) memory per window."""
    def __init__(self, window_seconds: float = 300, bucket_seconds: float = 1):
        self.window = window_seconds
        self.bucket_size = bucket_seconds
        self.num_buckets = int(window_seconds / bucket_seconds)
        self.buckets = deque(maxlen=self.num_buckets)
        self.current_bucket_time = 0
        self.current_count = 0

    def _advance_time(self):
        now = time.time()
        current = int(now / self.bucket_size)
        if current > self.current_bucket_time:
            # Flush current bucket and advance
            if self.current_bucket_time > 0:
                self.buckets.append(self.current_count)
            # Fill gaps with zeros
            gaps = min(current - self.current_bucket_time - 1, self.num_buckets)
            for _ in range(gaps):
                self.buckets.append(0)
            self.current_bucket_time = current
            self.current_count = 0

    def increment(self, count=1):
        self._advance_time()
        self.current_count += count

    @property
    def total(self) -> int:
        self._advance_time()
        return sum(self.buckets) + self.current_count

    @property
    def rate_per_second(self) -> float:
        return self.total / self.window

# Usage: detect anomalies
error_window = SlidingWindowCounter(window_seconds=300)  # 5-min window
request_window = SlidingWindowCounter(window_seconds=300)

def on_request(status_code):
    request_window.increment()
    if status_code >= 500:
        error_window.increment()
    # Alert if error rate exceeds 5%
    if request_window.total > 100:  # minimum sample size
        error_rate = error_window.total / request_window.total
        if error_rate > 0.05:
            trigger_alert(f"Error rate {error_rate:.1%} exceeds 5% threshold")

Pattern 7: Config File Watcher with Hot Reload

"""Watch config file for changes and hot-reload without restart."""
import os, time, json, hashlib, threading, logging

logger = logging.getLogger(__name__)

class ConfigWatcher:
    def __init__(self, config_path, on_change_callback, poll_interval=5):
        self.path = config_path
        self.callback = on_change_callback
        self.poll_interval = poll_interval
        self.last_hash = None
        self.config = {}
        self.lock = threading.RLock()
        self._load()
        self._start_watcher()

    def _compute_hash(self):
        try:
            with open(self.path, 'rb') as f:
                return hashlib.sha256(f.read()).hexdigest()
        except FileNotFoundError:
            return None

    def _load(self):
        try:
            with open(self.path, 'r') as f:
                new_config = json.load(f)
            with self.lock:
                self.config = new_config
                self.last_hash = self._compute_hash()
            logger.info(f"Config loaded: {self.path}")
            return True
        except (json.JSONDecodeError, FileNotFoundError) as e:
            logger.error(f"Config load failed: {e}")
            return False  # keep old config on parse failure

    def _start_watcher(self):
        t = threading.Thread(target=self._watch_loop, daemon=True)
        t.start()

    def _watch_loop(self):
        while True:
            time.sleep(self.poll_interval)
            current_hash = self._compute_hash()
            if current_hash and current_hash != self.last_hash:
                logger.info("Config change detected, reloading...")
                if self._load():
                    self.callback(self.config)

    def get(self, key, default=None):
        with self.lock:
            return self.config.get(key, default)

Section 04 — Linux & Systems

Linux Internals & Kernel Reasoning

Google SRE interviews test mechanistic understanding of Linux — not command memorization. They want to know WHY you chose a command, WHAT signal you expect, and HOW the kernel handles the operation.

Process Management

Process states: Running (R), Sleeping (S), Uninterruptible Sleep (D), Zombie (Z), Stopped (T). The D-state is critical — it means the process is waiting on I/O and cannot be killed (even with SIGKILL).

Key concepts: fork/exec model, copy-on-write, process groups, sessions, orphan processes, zombie reaping.

Interview hot topic: "Why would a process be stuck in D-state?" — Waiting on uninterruptible I/O (disk, NFS mount). Indicates storage/network issue, not a process bug.

Memory Management

Virtual memory: Each process gets its own virtual address space. The kernel maps virtual pages to physical frames via page tables. The MMU handles translation.

Key concepts: Page cache, slab allocator, OOM killer, memory reclaim, swap, RSS vs VSZ, memory-mapped files, huge pages.

Interview hot topic: "The OOM killer triggered — how do you investigate?" — Check dmesg for OOM messages, identify the killed process, check /proc/meminfo for memory pressure.

File Systems & I/O

Inodes: Store metadata (permissions, timestamps, block pointers) — NOT the filename. Filenames live in directory entries that map name → inode number.

Key concepts: VFS layer, page cache, direct I/O, write-back vs write-through, journaling, inode exhaustion, file descriptors.

Interview hot topic: "Disk is full but df shows space" — Deleted files still held open by processes. Use lsof +D /path to find them.

Networking Internals

TCP handshake: SYN → SYN-ACK → ACK. The SYN backlog queue is finite — if saturated, new connections are dropped.

Key concepts: Socket states, TIME_WAIT, connection pooling, TCP tuning (backlog, keepalive), DNS resolution, iptables/nftables, network namespaces.

Interview hot topic: "Intermittent connection timeouts" — Check SYN backlog (ss -s), check application thread pool, check DNS TTL, check for packet loss (mtr).

Scheduling & CPU

CFS (Completely Fair Scheduler): Uses a red-black tree of tasks sorted by virtual runtime. Tasks with less CPU time get priority.

Key concepts: Context switches, CPU affinity, cgroups CPU limits, load average (1/5/15 min), runqueue depth, CPU steal time (VMs).

Interview hot topic: "High load average but low CPU%" — Many processes in D-state (I/O wait) inflate load average without consuming CPU.

Cgroups & Containers

Cgroups v2: Hierarchical resource control — CPU, memory, I/O, PIDs. Prevents noisy neighbor problems.

Key concepts: Memory limits (hard/soft), CPU shares vs quotas, I/O weight, PID limits, OOM within cgroup, namespace types (PID, NET, MNT, UTS, IPC, USER).

Interview hot topic: "How do cgroups prevent noisy neighbors?" — Memory cgroup enforces hard limits; CPU cgroup uses CFS bandwidth throttling.

Section 05 — Linux Deep Dive

Advanced Kernel Concepts

These topics separate L4 from L5 candidates. Google expects Senior SREs to reason about kernel behavior mechanistically — not just know commands, but understand the data structures and algorithms the kernel uses.

Memory Reclaim & the Page Cache

The Linux kernel aggressively uses free RAM as page cache — caching file data for faster subsequent reads. This is why free shows little "free" memory on a healthy system. The key insight: used memory ≠ unavailable memory.

Page Cache Lifecycle

1. Process reads file → kernel loads pages into page cache
2. Subsequent reads hit cache (no disk I/O)
3. Under memory pressure, kernel reclaims clean pages (can be re-read from disk)
4. Dirty pages must be written back to disk before reclaim
5. If reclaim can't free enough → OOM killer activates

Key /proc/meminfo Fields

MemAvailable — what's actually available (free + reclaimable cache)
Cached — page cache size
Dirty — pages modified but not yet written to disk
Slab — kernel data structure caches (dentries, inodes)
SReclaimable — slab memory that CAN be freed
SUnreclaim — slab memory that CANNOT be freed

The OOM Killer Decision
The OOM killer selects victims using oom_score (0-1000). Score is based on: RSS (memory usage), oom_score_adj (admin override, -1000 to +1000), and whether the process is privileged. Check with: cat /proc/PID/oom_score. Set oom_score_adj = -1000 to make a process unkillable by OOM.

eBPF: The Modern SRE Superpower

eBPF (Extended Berkeley Packet Filter) allows running sandboxed programs inside the Linux kernel without modifying kernel source or loading kernel modules. It's revolutionizing observability, networking, and security.

What eBPF Can Do

Observability: Trace any kernel function, measure latency histograms, count events — all with near-zero overhead.
Networking: XDP (eXpress Data Path) processes packets before they reach the network stack — 10x faster than iptables.
Security: Seccomp-BPF filters syscalls. Runtime security monitoring without kernel modules.

Key eBPF Tools for SRE

bpftrace — one-liner tracing (like awk for kernel events)
bcc — Python + C toolkit for complex traces
tcplife — trace TCP connection lifetimes
biolatency — histogram of block I/O latency

Section 06 — Linux Troubleshooting Labs

Real-World Linux Scenarios

These are the exact types of scenarios you'll face in the troubleshooting round. Each one teaches a different kernel concept and debugging methodology. Practice thinking through them out loud.

Scenario 1: The Phantom Disk Usage

ALERT: /data partition at 100% — but du -sh /data shows only 60% used

Diagnosis Flow:

Hypothesis: Deleted files are still held open by running processes. When a file is deleted while a process has it open, the directory entry is removed (so du doesn't see it) but the inode and data blocks remain allocated until the file descriptor is closed.

# Step 1: Confirm the discrepancy
$ df -h /data        # shows 100% used
$ du -sh /data       # shows 60% — where's the other 40%?

# Step 2: Find deleted-but-open files
$ lsof +L1 /data     # +L1 = files with link count < 1 (deleted)
COMMAND   PID  USER   FD   TYPE DEVICE SIZE/OFF NLINKS NODE NAME
java     1234  app   45w   REG  253,1  42G       0    ... /data/app.log (deleted)

# Step 3: Recover space without killing the process
$ : > /proc/1234/fd/45   # truncate the file via its fd
# OR restart the process to release the fd

Root Cause & Prevention:

The application was writing to a log file that logrotate deleted, but the app never reopened its log file. Fix: configure the app to handle SIGHUP (reopen logs), or use copytruncate in logrotate config instead of create.

Scenario 2: Fork Bomb Recovery

ALERT: Server unresponsive. SSH connections timing out. Console shows "fork: retry: Resource temporarily unavailable"

Diagnosis Flow:

What happened: A fork bomb (e.g., :(){ :|:& };:) exhausted the system's PID limit. Every fork() call fails because the process table is full.

# If you can get a console (not SSH — SSH needs to fork):

# Step 1: Kill the fork bomb (use exec to avoid forking)
$ exec killall -9 -u baduser  # kill all processes by the offending user

# Step 2: If even that fails, use echo to /proc
$ echo 1 > /proc/sys/kernel/sysrq  # enable sysrq
$ echo i > /proc/sysrq-trigger     # kill all processes except init

# Prevention: Set PID limits
$ ulimit -u 1000                     # per-user process limit
$ echo "baduser hard nproc 500" >> /etc/security/limits.conf

# Better: Use cgroups PID controller
$ echo 500 > /sys/fs/cgroup/user.slice/user-1000.slice/pids.max

Prevention:

1. Set nproc limits in /etc/security/limits.conf
2. Use cgroups v2 PID controller for per-user/per-service limits
3. In Kubernetes: set spec.containers[].resources.limits.pids
4. Monitor PID usage: cat /proc/sys/kernel/pid_max vs ls /proc | grep -c '^[0-9]'

Scenario 3: TCP TIME_WAIT Exhaustion

ALERT: Service returning "Cannot assign requested address" errors. Connection rate dropping.

Diagnosis Flow:

# Step 1: Check connection states
$ ss -s
TCP:   65432 (estab 200, closed 100, orphaned 5, timewait 64000)
# ^ 64,000 TIME_WAIT connections! Ephemeral port range is only 28,232 ports

# Step 2: Understand the problem
# Ephemeral port range: 32768-60999 = 28,232 ports
# Each TIME_WAIT holds a port for 60 seconds
# If creating >470 connections/sec to same dest, ports exhaust

# Step 3: Immediate mitigation
$ sysctl -w net.ipv4.tcp_tw_reuse=1  # reuse TIME_WAIT for new outbound

# Step 4: Long-term fixes
$ sysctl -w net.ipv4.ip_local_port_range="1024 65535"  # expand range
# AND: implement connection pooling in the application
# AND: use HTTP keepalive to reuse connections

Key Insight:

TIME_WAIT exists to prevent old packets from a previous connection being accepted by a new connection on the same port. It's a safety mechanism, not a bug. The real fix is connection pooling — reuse connections instead of creating new ones for every request. tcp_tw_reuse is safe for outbound connections; never use the deprecated tcp_tw_recycle.

Scenario 4: Memory Leak in a Long-Running Service

ALERT: Service RSS growing 100MB/hour. OOM kill expected in ~6 hours.

Diagnosis Flow:

# Step 1: Confirm the leak pattern
$ while true; do
    echo "$(date) $(grep VmRSS /proc/$(pidof myservice)/status)"
    sleep 60
  done
# Watch RSS grow over time

# Step 2: Check if it's heap or mmap growth
$ cat /proc/PID/smaps_rollup
Rss:             4500000 kB
Pss:             4200000 kB
Shared_Clean:     200000 kB
Private_Dirty:   4000000 kB   # ← this is the leak (heap/anon)

# Step 3: For Python services — check with tracemalloc
# Add to the service code:
import tracemalloc
tracemalloc.start()
# ... later ...
snapshot = tracemalloc.take_snapshot()
top_stats = snapshot.statistics('lineno')
for stat in top_stats[:10]:
    print(stat)

# Step 4: For C/Go services — use pmap for mapping analysis
$ pmap -x PID | sort -k3 -n -r | head -20
# Look for large anonymous mappings growing over time

# Step 5: Check for file descriptor leaks (each fd = memory)
$ ls /proc/PID/fd | wc -l   # growing fd count = leak
$ cat /proc/PID/fdinfo/N    # details of specific fd

Common Causes:

1. Unbounded caches/maps — data structures that grow but never evict
2. Goroutine/thread leaks — spawning workers that never terminate
3. Connection leaks — HTTP clients not closing response bodies
4. Event listener accumulation — registering callbacks without deregistering
5. Slab cache growth — kernel dentries/inodes (check slabtop)

Scenario 5: The Mysterious Latency Spike

ALERT: p99 latency jumped from 50ms to 2000ms. p50 unchanged at 20ms. CPU/memory normal.

Diagnosis Flow:

# Key insight: p99 spike with normal p50 = a SUBSET of requests are slow
# This is NOT a general overload — it's targeted

# Step 1: Check for I/O latency (the silent killer)
$ iostat -x 1 3
# If await is high → disk I/O contention

# Step 2: Check for GC pauses (JVM/Go/Python)
$ grep "GC pause" /var/log/app/gc.log | tail -20
# GC pauses affect a random subset of requests

# Step 3: Check for lock contention
$ perf top -p PID
# If you see futex_wait or pthread_mutex_lock dominating → lock contention

# Step 4: Check for DNS resolution delays
$ strace -p PID -e trace=connect,sendto -T 2>&1 | grep -i "dns\|53"
# DNS timeouts cause exactly this pattern: most requests fast, some 2s+ slow

# Step 5: Check for connection pool exhaustion
$ ss -tnp | grep PID | wc -l   # how many connections does the process have?
# If at pool max → requests queue waiting for a connection

# Step 6: Check for CPU throttling (cgroups)
$ cat /sys/fs/cgroup/cpu/docker/CONTAINER_ID/cpu.stat
nr_throttled 15234
throttled_time 89000000000  # 89 seconds of throttling!

The Pattern:

"p99 spike with normal p50" almost always means one of: GC pauses, DNS timeouts, lock contention, connection pool exhaustion, or CPU throttling in cgroups. The key diagnostic is that it's a bimodal distribution — most requests are fine, a few are very slow.

Scenario 6: Inode Exhaustion

ALERT: "No space left on device" but df -h shows 40% free space

Diagnosis Flow:

# Step 1: Check inodes separately from blocks
$ df -i
Filesystem     Inodes   IUsed   IFree IUse% Mounted on
/dev/sda1      5242880  5242880     0  100%  /

# Step 2: Find where the inodes are consumed
$ find / -xdev -printf '%h\n' | sort | uniq -c | sort -rn | head -10
 2500000 /var/spool/mail/queue
  500000 /tmp/sessions
# ^ millions of tiny files in one directory

# Step 3: Identify the culprit
# Common causes: mail queue buildup, PHP session files,
# container overlay layers, package manager cache

# Step 4: Clean up (carefully!)
$ find /var/spool/mail/queue -type f -mtime +7 -delete
# Delete files older than 7 days

# Prevention: Monitor inode usage in your alerting
# Set up tmpwatch/systemd-tmpfiles for temp directories

Key Insight:

Each file (even 0 bytes) consumes one inode. Filesystems have a fixed number of inodes set at creation time. Millions of tiny files (session files, cache entries, mail queue) can exhaust inodes while leaving plenty of block space. Prevention: use mkfs -N to set inode count at filesystem creation, or use XFS which allocates inodes dynamically.

Scenario 7: CPU Steal Time in VMs

ALERT: Application latency increased 3x. top shows 40% steal time (%st).

Diagnosis Flow:

# Step 1: Confirm steal time
$ mpstat 1 5
%usr  %sys  %iowait  %steal  %idle
 25    10      5       40      20
# 40% steal = the hypervisor is taking 40% of our CPU time

# Step 2: Understand what steal means
# Steal time = time the vCPU wanted to run but the hypervisor
# scheduled a different VM's vCPU on the physical core instead.
# It's the "noisy neighbor" problem in shared hosting.

# Step 3: Verify it's not our fault
$ cat /proc/cpuinfo | grep "model name"  # check CPU type
$ nproc                                   # check vCPU count
$ uptime                                  # check load average
# If our load is low but steal is high → noisy neighbor

# Step 4: Mitigation options
# - Request migration to a different host (cloud provider)
# - Switch to dedicated/bare-metal instances
# - Use CPU pinning (if available)
# - Scale horizontally to reduce per-instance load

Key Insight:

CPU steal time is invisible to the application — it just sees slower execution. It's a hypervisor-level problem you can't fix from inside the VM. In cloud environments, this is the #1 reason to use dedicated instances for latency-sensitive workloads. Always monitor %st in your dashboards.

Scenario 8: Network Partition Debugging

ALERT: Service A can't reach Service B. Both services report healthy locally.

Layer-by-Layer Diagnosis:

# Step 1: DNS — does the name resolve correctly?
$ dig service-b.internal +short
10.0.5.42
$ dig @8.8.8.8 service-b.internal +short  # compare with external DNS

# Step 2: L3 — can we reach the IP at all?
$ ping -c 3 10.0.5.42
# If no response: routing issue, firewall, or host down

# Step 3: L4 — can we establish TCP connection?
$ nc -zv 10.0.5.42 8080 -w 5
# If timeout: firewall blocking the port, service not listening

# Step 4: Path analysis — where are packets being dropped?
$ mtr -n --report 10.0.5.42
# Shows packet loss at each hop

# Step 5: Check local firewall rules
$ iptables -L -n -v | grep -i drop
$ nft list ruleset | grep drop

# Step 6: Check for network namespace issues (containers)
$ ip netns list
$ nsenter -t PID -n ss -lntp  # check from inside container's netns

# Step 7: Packet capture (last resort)
$ tcpdump -i any host 10.0.5.42 and port 8080 -nn -c 20
# Look for: SYN without SYN-ACK (firewall), RST (connection refused),
# retransmissions (packet loss)

Common Causes:

1. Security group/firewall rule change — most common in cloud
2. DNS propagation delay — service moved but DNS cached old IP
3. Network policy (Kubernetes) — new policy blocking traffic
4. MTU mismatch — packets too large being silently dropped
5. ARP cache poisoning — stale MAC address in ARP table

Scenario 9: Kernel Soft Lockup

ALERT: dmesg shows "BUG: soft lockup - CPU#3 stuck for 22s!"

Diagnosis Flow:

# Step 1: Check dmesg for the full message
$ dmesg -T | grep -A 20 "soft lockup"
# Shows which CPU, which process, and the kernel stack trace

# What is a soft lockup?
# A CPU has been running kernel code for >20s without scheduling.
# The watchdog timer fires because the CPU hasn't serviced interrupts.

# Step 2: Identify the cause from the stack trace
# Common patterns:
# - spin_lock → lock contention in kernel
# - nfs/cifs → network filesystem hung
# - ext4/xfs → filesystem corruption or I/O stall
# - virtio → hypervisor/VM issue

# Step 3: Check if it's a hardware issue
$ mcelog --client  # check for hardware errors (ECC, CPU)
$ lspci -v | grep -i error

# Step 4: Check I/O subsystem
$ cat /sys/block/sda/device/timeout  # SCSI timeout
$ dmesg | grep -i "reset\|timeout\|error" | tail -20

# Mitigation:
# - If NFS: check NFS server health, consider soft mount
# - If disk: check disk health (smartctl), replace if failing
# - If VM: check hypervisor logs, consider live migration

Key Insight:

Soft lockups are kernel-level issues — they indicate a CPU is stuck in kernel code. Unlike hard lockups (CPU completely frozen), soft lockups still allow the watchdog to fire. They're usually caused by: storage I/O stalls, kernel bugs, hardware issues, or hypervisor problems. The kernel stack trace in dmesg is your primary diagnostic tool.

Scenario 10: DNS Resolution Failures

ALERT: Intermittent "Name or service not known" errors. ~5% of requests failing.

Diagnosis Flow:

$ dmesg | grep "nf_conntrack: table full"

1. UDP receive buffer overflow — resolver can't keep up with query rate
2. Conntrack table full — especially in Kubernetes with many short-lived DNS queries
3. ndots:5 in resolv.conf — Kubernetes default causes 5x DNS queries per lookup
4. DNS server overloaded — too many pods querying CoreDNS
5. Race condition in /etc/resolv.conf — DHCP renewal rewriting the file

Scenario: Cascading Failure After Dependency Timeout

ALERT: Service A latency 10x normal. Service B (dependency) returned to normal 5 minutes ago.

What's happening:

Service B had a brief outage. Service A's requests to B timed out, filling A's thread pool. Even after B recovered, A's thread pool is still exhausted because backed-up requests are still being processed.

# Diagnosis:
$ ss -tnp | grep SERVICE_A_PID | wc -l  # connection count
$ curl localhost:8080/metrics | grep thread_pool
# thread_pool_active: 200/200 (saturated!)
# thread_pool_queue: 15000 (backed up!)

# Fix: Shed the queued load
# 1. Restart Service A (clears the queue)
# 2. OR: if it has a queue-drop endpoint, trigger it

# Prevention: Circuit breaker + request deadline propagation
# If a request has been waiting > its deadline, drop it immediately

Key Lesson:

This is the "thundering herd" / "retry storm" pattern. Prevention requires: circuit breakers (stop calling failing dependencies), request deadlines (drop requests that have already timed out from the user's perspective), and load shedding (reject new requests when queue depth exceeds threshold).

Section 10 — Test Yourself

Rapid-Fire Quiz (25 Questions)

Test your knowledge across all interview dimensions. Each question mirrors real Google SRE interview patterns.

Q1: A process is in D-state (uninterruptible sleep). What does this mean?
D-state means the process is in uninterruptible sleep, typically waiting on disk I/O or an NFS mount. It cannot be killed because the kernel needs the I/O operation to complete to maintain filesystem consistency.
Q2: Which approach is correct for processing a 50GB log file with bounded memory?
Using for line in f treats the file as an iterator, reading one line at a time with O(1) memory. All other options load the entire file into memory.
Q3: 99.9% availability SLO — how much downtime per month?
99.9% = 0.1% downtime. 30 days × 24h × 60min = 43,200 min. 0.1% of 43,200 = 43.2 minutes. This is your monthly error budget.
Q4: What should you do FIRST when told "the site is down" in a troubleshooting interview?
The "Mitigation-First" mindset is the #1 signal. Assess impact, then mitigate (drain, rollback, failover). Only after stabilization do you diagnose root cause.
Q5: What distinguishes "process" from "experience" questions in the Googleyness round?
Experience: "Tell me about a time..." → deep STAR story. Process: "How do you approach..." → framework first, then brief example. Answering both with just a story leaves signal on the table.
Q6: High load average but low CPU utilization. Most likely cause?
Linux load average counts ALL processes in the run queue, including D-state. Many processes blocked on I/O inflate load average without consuming CPU. Verify with vmstat (wa column).
Q7: Replicate 5 PB with 4-hour RTO. Minimum bandwidth needed?
5 PB = 40,000 Tb. 40,000 Tb / 4h = 10,000 Tb/h ≈ 2.8 Tbps. This exceeds standard links — the design must use incremental replication or accept longer RTO.
Q8: What is "toil" in SRE?
Google defines toil as manual, repetitive, automatable, tactical, without enduring value, and scaling linearly with growth. Target: keep below 50% of SRE time.
Q9: Best concurrency pattern for checking 100K servers with rate limiting?
asyncio is ideal for I/O-bound work. Semaphore limits concurrent requests, token bucket enforces rate. 100K threads would exhaust OS resources.
Q10: What does "blamelessness" mean in SRE behavioral stories?
Frame outages as system failures: "The pipeline lacked a canary stage" not "Bob pushed bad code." Focus on what the system allowed to happen and what you changed to prevent recurrence.
Q11: df -h shows 40% free but you get "No space left on device". What's wrong?
Each file consumes one inode regardless of size. Millions of tiny files can exhaust inodes while leaving block space available. Always check both df -h AND df -i.
Q12: What is the purpose of exponential backoff with FULL jitter in retry logic?
Full jitter (random between 0 and calculated delay) decorrelates retries across multiple clients. Without jitter, all clients retry at the same time, creating a "thundering herd" that overwhelms the recovering service.
Q13: You see 40% CPU steal time (%st) in top. What does this indicate?
CPU steal time means the hypervisor took CPU time away from your VM to give to other VMs on the same physical host. It's the "noisy neighbor" problem — you can't fix it from inside the VM. Solution: dedicated instances or migration.
Q14: A circuit breaker is in "OPEN" state. What happens to incoming requests?
When a circuit breaker is OPEN, all requests fail immediately (fast-fail) without attempting to call the failing dependency. This prevents resource exhaustion from timeouts and gives the dependency time to recover. After a cooldown period, it transitions to HALF-OPEN to test recovery.
Q15: Why does tail -f need to check the file's inode number?
When logrotate rotates a file, it typically renames the old file and creates a new one at the same path. The new file has a different inode number. By checking the inode, tail -f detects this rotation and reopens the new file. Without this check, it would keep reading the old (renamed) file.
Q16: What is PSI (Pressure Stall Information) in Linux?
PSI (/proc/pressure/) quantifies resource contention. It shows "some" (at least one task stalled) and "full" (ALL tasks stalled) percentages for CPU, memory, and I/O. It's more actionable than load average because it directly measures impact on task execution.
Q17: p99 latency spiked but p50 is unchanged. Which is LEAST likely the cause?
General CPU overload would affect ALL requests, increasing both p50 and p99. The "p99 spike with normal p50" pattern indicates a bimodal distribution — most requests are fine, a few are very slow. This points to intermittent issues like GC pauses, DNS timeouts, or pool exhaustion that only affect a subset of requests.
Q18: What does net.ipv4.tcp_tw_reuse=1 do?
tcp_tw_reuse allows the kernel to reuse a socket in TIME_WAIT state for a new outbound connection to the same IP:port, using TCP timestamps to ensure safety. It's safe for clients. Note: tcp_tw_recycle (now removed) was dangerous and should never be used.
Q19: What's the key advantage of eBPF over strace for production tracing?
strace uses ptrace() which context-switches on every syscall, causing 2-5x overhead. eBPF programs run inside the kernel with JIT compilation and near-zero overhead. This makes eBPF safe for production tracing, while strace should be a last resort.
Q20: In a connection pool, what should happen when a connection throws an exception?
A connection that threw an exception may be in an inconsistent state (partial response, broken protocol state). It should be closed and discarded, not returned to the pool. The pool will create a fresh connection when needed. Returning broken connections causes cascading errors for subsequent users of that connection.
Q21: What does %util in iostat mean for NVMe drives?
%util measures the percentage of time at least one I/O was in progress. For single-queue devices (HDD), 100% = saturated. But NVMe drives have deep parallel queues — they can be at 100% util while handling 10x more IOPS easily. For NVMe, focus on await and queue depth instead.
Q22: A service has CLOSE_WAIT connections accumulating. Where is the bug?
CLOSE_WAIT means: the remote side sent FIN (closed their end), but the local application hasn't called close() yet. This is always a local application bug — typically a missing close() in an error handling path, or a connection leak where the application lost the reference to the socket.
Q23: What is the correct way to handle SIGTERM in a production service?
Graceful shutdown: stop accepting new work (drain), complete in-flight requests (within a timeout), flush metrics/logs, close connections cleanly, then exit. Kubernetes sends SIGTERM 30s before SIGKILL — you have 30s to shut down gracefully.
Q24: What causes "conntrack table full" errors in Kubernetes?
Linux conntrack tracks every connection for NAT/firewall purposes. In Kubernetes, every DNS query (UDP), service connection, and pod-to-pod communication creates a conntrack entry. High-churn workloads (many short-lived connections) can exhaust the table. Fix: increase nf_conntrack_max or reduce connection churn.
Q25: What does the OOM killer use to select which process to kill?
The OOM killer calculates oom_score (0-1000) based primarily on RSS (memory usage) and the admin-configurable oom_score_adj (-1000 to +1000). The process with the highest score gets killed. Set oom_score_adj=-1000 to protect critical processes.

Coding Round

How would you parse a 50GB log file with 512MB memory limit? +

"I'd treat the file as a stream. In Python, iterating over a file object reads one line at a time with O(1) memory. For aggregation, I'd use a Counter bounded by unique keys, not file size. For top-N results, I'd use heapq.nlargest which is O(n log k). I'd add error handling for malformed lines — skip them with a warning rather than crashing. I'd also set a larger buffer size for I/O performance. The key insight is separating the streaming read from the bounded aggregation."

Implement a rate limiter. Walk me through your design. +

"I'd use a token bucket because it handles burst traffic naturally. Two parameters: rate (tokens/second) and burst (max tokens). Tokens refill lazily — on each acquire(), I calculate elapsed time and add proportional tokens capped at burst. This avoids a background thread. For thread safety, I'd wrap refill-and-consume in a lock. The acquire() method returns True/False for non-blocking use, and wait() blocks until tokens are available. The alternative is a sliding window counter — simpler but doesn't handle bursts as gracefully."

Write a script that implements tail -f with log rotation support. +

"I'd open the file, seek to the end, then enter a polling loop. On each iteration, I try readline(). If there's data, yield it. If not, I check for two conditions: rotation and truncation. For rotation, I compare the current file's inode (via os.stat) against the inode of my open file descriptor (via os.fstat). If they differ, the file was rotated — I close and reopen. For truncation, I check if the file size is less than my current position — if so, seek to 0. This mirrors how the real tail -F works. I'd use a configurable poll interval, defaulting to 100ms."

Design a connection pool with health checking. What are the key considerations? +

"The pool needs four key behaviors. First, bounded size — a max_size parameter prevents resource exhaustion. Second, health checking — a background thread periodically tests idle connections and evicts unhealthy ones. Third, safe checkout/return — I'd use a context manager so connections are always returned even on exceptions. Critically, if a connection throws an exception during use, I close and discard it rather than returning it to the pool — it might be in an inconsistent state. Fourth, idle timeout — connections unused for too long get evicted because the server may have closed them. I'd use a queue internally for thread-safe FIFO access."

How would you detect anomalies in a stream of events from millions of hosts? +

"I'd use a sliding window counter per host, implemented with a deque of time-bucketed counts. Each bucket represents one second, and the deque holds 300 buckets for a 5-minute window. When an event arrives, I advance the time, increment the current bucket, and compute the rate. For fleet-wide comparison, I'd maintain a global sliding window for the average rate. An anomaly is when a host's rate exceeds 3x the fleet average. The memory is bounded: one deque per host with fixed-size buckets. For millions of hosts, I might use approximate data structures like Count-Min Sketch if exact counts aren't needed."

Linux Internals

Explain what happens in the kernel when you run ls in a terminal. +

"The shell calls fork() to create a child process — this uses copy-on-write so it's fast, no memory is actually copied yet. The child then calls execve('/bin/ls', ...) which replaces the process image: the kernel loads the ELF binary, sets up the new virtual address space, maps shared libraries via the dynamic linker, and jumps to the entry point. ls then calls opendir() which becomes a getdents64 syscall — the kernel reads directory entries from the filesystem's inode. For each entry, ls may call stat() to get metadata. The results are written to stdout via write() syscalls. Finally, ls calls exit(), the kernel cleans up resources, and sends SIGCHLD to the parent shell, which calls wait() to reap the child."

A service is running in a container and getting OOM-killed. How do you investigate? +

"First, I'd check dmesg for the OOM message — it tells me which cgroup triggered the kill and the memory state at the time. Then I'd check the cgroup's memory limit: cat /sys/fs/cgroup/memory/docker/CONTAINER_ID/memory.limit_in_bytes. I'd compare this against the actual usage over time using memory.usage_in_bytes. To understand what's consuming memory, I'd look at memory.stat which breaks down cache vs RSS vs swap. If RSS is growing unboundedly, it's a leak in the application. If cache is high, the container might be doing heavy file I/O and the page cache is filling the cgroup's limit — in that case, the fix might be increasing the limit since cache is reclaimable. I'd also check if the application has internal metrics showing heap size or connection count growth."

What is eBPF and when would you use it instead of traditional tools? +

"eBPF lets you run sandboxed programs inside the Linux kernel without modifying kernel source or loading modules. The programs are verified for safety by the kernel before execution and JIT-compiled for performance. I'd use it instead of strace when I need production-safe tracing — strace has 2-5x overhead via ptrace, while eBPF has near-zero overhead. Specific use cases: biolatency for disk I/O histograms, tcplife for connection lifetime analysis, execsnoop for tracking process execution, and custom bpftrace one-liners for ad-hoc investigation. For example, if I need to trace which files a service is opening in production without impacting latency, I'd use opensnoop (eBPF-based) rather than strace."

Explain the difference between free memory and available memory in Linux. +

"Free memory is completely unused RAM — nothing is stored there. Available memory is free memory PLUS reclaimable memory — primarily clean page cache pages that the kernel can evict instantly if needed. On a healthy Linux system, free memory is often very low because the kernel aggressively uses spare RAM as page cache to speed up file access. This is not a problem — it's optimal behavior. The metric that matters is MemAvailable in /proc/meminfo, not MemFree. A system with 100MB free but 8GB available is healthy. A system with 100MB available is under memory pressure. The old 'free' command confused people by showing low free memory; modern versions show 'available' which is what you should monitor."

How would you debug a process that's consuming 100% of one CPU core? +

"First, I'd identify what the CPU time is spent on using top — is it user time (us) or system time (sy)? If user time, the process is doing computation or spinning in a loop. If system time, it's making excessive syscalls. For user-time issues, I'd use perf top -p PID to see which functions are hot — this shows me whether it's a busy loop, an inefficient algorithm, or lock spinning (futex). For system-time issues, I'd use strace -c -p PID to get a syscall summary — maybe it's doing millions of tiny reads or futex calls. If it's a known application, I'd check for CPU profiling endpoints (like Go's pprof or Java's async-profiler). The key is distinguishing between 'doing useful work fast' and 'spinning wastefully' — the former might just need more capacity, the latter is a bug."

Troubleshooting Round

The site is slow. CPU is normal. Walk me through your diagnosis. +

"CPU-normal slowness points to either I/O, network, or lock contention. I'd start with vmstat 1 5 — if the 'wa' column is high, it's disk I/O. If 'si/so' are non-zero, it's swap pressure. Next, ss -s to check connection states — a large SYN backlog or many TIME_WAIT connections indicates network issues. Then iostat -x 1 3 — if await is high, disk I/O is the bottleneck. If none of these show issues, I'd check for lock contention with perf top looking for futex_wait. I'd also check if the application has a thread pool that's exhausted — curl the metrics endpoint for queue depth. Each step gives me a specific signal that narrows the hypothesis space."

p99 latency doubled after a deploy. What do you do? +

"First, assess: is this within error budget? If not, I'd recommend immediate rollback while we investigate — mitigation first. If we can afford to investigate, I'd correlate: diff the release for new dependencies or config changes. Check canary metrics if available — compare canary vs baseline latency distribution. Determine if the increase is uniform or concentrated in specific endpoints or regions. Pull distributed traces for slow requests to see where time is spent. Common causes after deploy: new dependency with higher latency, missing index on a new query, increased logging volume, or a concurrency regression. The rollback gives us time to investigate without burning error budget."

A service can't reach its database. Diagnose layer by layer. +

"I'd go bottom-up through the network stack. Layer 1 — DNS: dig db-host.internal — does it resolve? Is the IP correct? Layer 2 — TCP: nc -zv db-host 5432 — can we establish a connection? If timeout, it's network/firewall. If connection refused, the DB isn't listening. Layer 3 — TLS: openssl s_client if encrypted — certificate valid? Not expired? Layer 4 — Auth: check credentials, IAM roles, pg_hba.conf. Layer 5 — Capacity: SELECT count(*) FROM pg_stat_activity — is the connection pool full? Layer 6 — Firewall: check security groups, network policies. I'd also check if this is affecting all instances or just one — that distinguishes between a DB-side issue and a network/config issue on one host."
